#!/usr/bin/env python3
"""Run synthetic SwiftData migration checks in a new directory on macOS/Xcode."""
import argparse
import hashlib
import json
import platform
import shutil
import subprocess
import sys
import tempfile
from pathlib import Path

FIRST = '11111111-1111-1111-1111-111111111111'
SECOND = '22222222-2222-2222-2222-222222222222'
EXPECTED = [
    {'id': FIRST, 'name': 'Crème brûlée', 'childrenIDs': [SECOND]},
    {'id': SECOND, 'name': 'Budget 2048', 'parentID': FIRST, 'childrenIDs': []},
]


def file_hash(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def tree_hashes(path):
    return {str(p.relative_to(path)): file_hash(p) for p in sorted(path.rglob('*')) if p.is_file()}


def execute(command, log, timeout=120):
    result = subprocess.run(command, capture_output=True, text=True, timeout=timeout)
    log.write_text(result.stdout + '\n--- stderr ---\n' + result.stderr)
    if result.returncode != 0:
        raise RuntimeError(f'Command failed ({result.returncode}); see {log}')
    return result.stdout


def check_snapshot(binary, operation, store, shape, log, expected_prior=None):
    result = json.loads(execute([str(binary), operation, str(store), shape], log))
    expected = EXPECTED if shape == 'records' else []
    version = '1.0.0' if operation == 'seed1' else '2.0.0' if operation in ('seed2', 'open2') else '3.0.0'
    if result.get('storeVersions') != [version]:
        raise RuntimeError(f'Unexpected schema-version metadata; see {log}')
    if not operation.startswith('seed') and result.get('priorVersions') != [expected_prior]:
        raise RuntimeError(f'Unexpected source schema metadata; see {log}')
    if result.get('records') != expected:
        raise RuntimeError(f'Unexpected persisted values/relationships; see {log}')
    return result


def run(work):
    source = Path(__file__).resolve().parents[1] / 'assets/MigrationProbe.swift'
    binary = work/'probe'
    report = {'schema_version': 1, 'source_sha256': file_hash(source),
              'runner_sha256': file_hash(Path(__file__)),
              'host': platform.platform(), 'cases': [], 'passed': False,
              'limitations': ['Synthetic schemas created using this SDK, not historical shipped application stores.',
                              'macOS SwiftData execution does not establish behavior on every supported iOS release.',
                              'No CloudKit, custom migration, uniqueness-conflict, or production-data verification.']}
    try:
        report['toolchain'] = execute(['xcrun', 'swift', '--version'], work/'toolchain.log').strip()
        execute(['xcrun', 'swiftc', '-swift-version', '6', '-warnings-as-errors', '-parse-as-library',
                 '-module-name', 'MigrationProbe', str(source), '-o', str(binary)], work/'compile.log')
        # Producer processes exit before we copy their entire store directories.
        fixtures = {}
        for version in (1, 2):
            for shape in ('records', 'empty'):
                folder = work/f'fixture-v{version}-{shape}'; folder.mkdir()
                check_snapshot(binary, f'seed{version}', folder/'store.sqlite', shape, work/f'seed-v{version}-{shape}.log')
                fixtures[version, shape] = (folder, tree_hashes(folder))
        for version, stages in ((1, ['open3']), (1, ['open2', 'open3']), (2, ['open3'])):
            for shape in ('records', 'empty'):
                name = f'v{version}-' + '-'.join(stages) + '-' + shape
                folder = work/name
                original, original_hashes = fixtures[version, shape]
                shutil.copytree(original, folder)
                prior = f'{version}.0.0'
                transitions = []
                for index, stage in enumerate(stages):
                    snapshot = check_snapshot(binary, stage, folder/'store.sqlite', shape, work/f'{name}-{index}.log', expected_prior=prior)
                    transitions.append({'from': prior, 'to': snapshot['storeVersions'][0]})
                    prior = snapshot['storeVersions'][0]
                # A new process reopens the migrated store, checking persisted rather than cached data.
                check_snapshot(binary, 'open3', folder/'store.sqlite', shape, work/f'{name}-reopen.log', expected_prior='3.0.0')
                if tree_hashes(original) != original_hashes:
                    raise RuntimeError('Pristine source fixture changed: ' + str(original))
                report['cases'].append({'name': name, 'passed': True, 'reopened': True,
                                        'source_unchanged': True, 'snapshot_matches_intended_seed': True, 'transitions': transitions})
        # Negative control: a rename implemented as a new defaulted property must not pass.
        mutant_source = work/'BrokenRename.swift'
        mutant_source.write_text(source.read_text().replace('@Attribute(originalName: "title") var name: String', 'var name: String = ""'))
        mutant_binary = work/'broken-probe'
        execute(['xcrun', 'swiftc', '-swift-version', '6', '-warnings-as-errors', '-parse-as-library',
                 '-module-name', 'MigrationProbe', str(mutant_source), '-o', str(mutant_binary)], work/'compile-negative.log')
        negative = work/'negative-rename'; shutil.copytree(fixtures[1, 'records'][0], negative)
        broken_snapshot = json.loads(execute([str(mutant_binary), 'open3', str(negative/'store.sqlite'), 'records'],
                                             work/'negative-rename.log'))
        expected_corruption = [dict(record, name='') for record in EXPECTED]
        if broken_snapshot.get('records') != expected_corruption or broken_snapshot.get('storeVersions') != ['3.0.0'] or broken_snapshot.get('priorVersions') != ['1.0.0']:
            raise RuntimeError('Negative control did not reproduce the expected silent rename data loss')
        report['negative_control'] = {'detected': True, 'migration_opened_successfully': True,
                                      'record_count_preserved': True, 'names_lost': True,
                                      'snapshot': broken_snapshot}
        report['passed'] = True
    except (OSError, ValueError, RuntimeError, subprocess.TimeoutExpired) as exc:
        report['error'] = str(exc)
    (work/'results.json').write_text(json.dumps(report, indent=2)+'\n')
    return report


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument('--output', type=Path, help='A new, nonexistent output directory; otherwise create a temporary one')
    args = p.parse_args()
    if sys.platform != 'darwin' or shutil.which('xcrun') is None:
        print('Requires macOS with Xcode and SwiftData support.', file=sys.stderr)
        return 2
    try:
        if args.output:
            work = args.output.resolve()
            work.mkdir(parents=False, exist_ok=False)
        else:
            work = Path(tempfile.mkdtemp(prefix='swiftdata-migration-probe-'))
    except OSError as exc:
        print(str(exc), file=sys.stderr)
        return 2
    report = run(work)
    print(json.dumps({'passed': report['passed'], 'results': str(work/'results.json'),
                      'cases': len(report['cases']), 'error': report.get('error')}, indent=2))
    return 0 if report['passed'] else 1


if __name__ == '__main__':
    sys.exit(main())
